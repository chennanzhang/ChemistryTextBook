\newcommand{\ClipedTube}[3][]{
  \begin{scope}[#1,rotate=#2]
    \draw[fill=darkgray](0,0.2)rectangle++(0.1,-1.0);
    \fill[lightgray!20](-#3/3*2,0.15)arc(90:270:0.15)--++(0.4,0)to[bend right=10](-#3/3*2,0.15);
    \fill[pattern={Dots[distance=0.6pt,radius=0.15pt,angle=40]},pattern color=darkgray](-#3/3*2,0.15)arc(90:270:0.15)--++(0.4,0)to[bend right=10](-#3/3*2,0.15);
    \draw[fill=cyan!10!white,fill opacity=0.4](-#3/3*2,0.15)--(#3/3,0.15)arc(-90:0:0.05)--(#3/3+0.05,-0.2)arc(0:90:0.05)--(-#3/3*2,-0.15)arc(270:90:0.15)--cycle;
    \foreach \x in {0,10,...,80}
    {
      \draw[ultra thin,darkgray](-0.4,{0.15*sin(\x)})--++(0.4,0);
      \draw[ultra thin,darkgray](-0.4,{-0.15*sin(\x)})--++(0.4,0);
    }
    \draw[fill=brown](0.02,0.22)rectangle++(0.1,-1.3);
    \draw[semithick,darkgray](0,-0.3)--(0.12,-0.35)arc(90:-90:0.01);
  \end{scope}
}
