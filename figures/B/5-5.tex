\newcommand{\retortstand}[4][]{
  \begin{scope}[#1]
    \draw[fill=darkgray](0,0)--++(#3,0)--++(0.05,-0.05)--++(0,-0.15)--++(-0.4,0)--++(0,0.05)--++(-#2-#3+0.7,0)--++(0,-0.05)--++(-0.4,0)--++(0,0.15)--++(0.05,0.05)--cycle;
    \draw(-#2-0.05,-0.05)--(#3+0.05,-0.05);
    \fill[left color=lightgray,right color=lightgray,middle color=white](-0.05,0)rectangle(0.05,#4);
    \fill[left color=gray,right color=gray,middle color=white](-0.07,0)rectangle(0.07,0.07);
  \end{scope}
}

\newcommand{\ConicalFlask}[3][]{
  \begin{scope}[#1]
    \fill[#3,rounded corners=5pt](0.65-#2/3,#2)--(0.65,0)--(-0.65,0)--(-0.65+#2/3,#2);
    \foreach \x/\y in {0/1.0,0.05/0.8}
    \draw[very thin,fill=cyan!20!white](\x,\y)to[in=180,out=-90]+(0,-0.1)to[in=-90,out=0]cycle;
    \draw[fill=cyan!5!white,fill opacity=0.2](-0.15,1.5)--(-0.15,1.8)arc(0:90:0.05)--(0.20,1.85)arc(90:180:0.05)[rounded corners=5pt]--(0.15,1.5)--(0.65,0)--(-0.65,0)--(-0.15,1.5)--cycle;
  \end{scope}
}
