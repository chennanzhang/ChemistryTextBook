\newcommand{\bulb}[1][]{
  \begin{scope}[#1]
    \draw[left color=brown,right color=brown,middle color=white](-0.3,0)--(-0.3,-0.1)arc(-180:0:0.3 and 0.1)--(0.3,0)arc(0:-180:0.3 and 0.1)--cycle;
    \draw[fill=brown!50](0,0)ellipse(0.3 and 0.1);
    \draw[left color=darkgray,right color=darkgray,middle color=white](-0.1,0.15)--(-0.1,0)arc(-180:0:0.1 and 0.025)--(0.1,0.15)arc(0:-180:0.1 and 0.025)--cycle;
    \draw[fill=lightgray!50](0,0.15)ellipse(0.1 and 0.025);
    \draw[very thin](-0.05,0.15)--(-0.05,0.3)--(-0.08,0.4)(0.05,0.15)--(0.05,0.3)--(0.08,0.4);
    \draw[very thin,decorate,decoration={zigzag,segment length=1pt,amplitude=0.5pt}](-0.08,0.4)--(0.08,0.4);
    \draw[fill=cyan!20!white,fill opacity=0.3](-0.1,0.15)--(-0.1,0.25)arc(240:-60:0.2)--(0.1,0.15);
    \draw[line cap=round,semithick](-0.2,0)--(-0.2,0.05)(0.2,0)--(0.2,0.05);
    \draw[line cap=round,very thick](-0.2,0.1)--(-0.2,0.05)(0.2,0.1)--(0.2,0.05);
  \end{scope}
}
