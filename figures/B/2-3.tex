\newcommand{\retortstand}[4][]{
  \begin{scope}[#1]
    \draw[fill=darkgray](0,0)--++(#3,0)--++(0.05,-0.05)--++(0,-0.15)--++(-0.4,0)--++(0,0.05)--++(-#2-#3+0.7,0)--++(0,-0.05)--++(-0.4,0)--++(0,0.15)--++(0.05,0.05)--cycle;
    \draw(-#2-0.05,-0.05)--(#3+0.05,-0.05);
    \fill[left color=lightgray,right color=lightgray,middle color=white](-0.05,0)rectangle(0.05,#4);
    \fill[left color=gray,right color=gray,middle color=white](-0.07,0)rectangle(0.07,0.07);
  \end{scope}
}
\newcommand{\beaker}[5][]{
  \begin{scope}[#1]
    \fill[#5,rounded corners=1mm](-#2/2,#4)--(-#2/2,0)--(#2/2,0)--(#2/2,#4);
    \draw[fill=cyan!10!white,fill opacity=0.3](#2/2+0.07,#3)arc(90:180:0.07)[rounded corners=1mm]--(#2/2,0)--(-#2/2,0)[rounded corners=0.5mm]--(-#2/2,#3-0.15)[sharp corners]--++(-0.1,0.1)--(-#2/2,#3)--cycle;
  \end{scope}
}
\newcommand{\rflask}[5][]{
  \begin{scope}[#1]
    \fill[#4,fill opacity=0.3](#3:0.5)arc(#3:-#3-180:0.5)--cycle;
    \fill[#5,fill opacity=0.3](#3:0.5)arc(#3:60:0.5)arc(240:180:0.2)--++(0,0.65)--++(-0.3,0)--++(0,-0.65)arc(0:-60:0.2)arc(120:180-#3:0.5);
    \draw(60:0.5)arc(60:-240:0.5)arc(-60:0:0.2)--++(0,0.5)coordinate(#2-bl)--++(0,0.1)arc(0:90:0.05)--++(0.4,0)arc(90:180:0.05)--++(0,-0.6)arc(180:240:0.2);
    \fill[left color=gray,right color=gray,middle color=white](#2-bl)rectangle++(0.3,0.25);
    \draw[semithick](-0.2,1.25)--(0.2,1.25);
  \end{scope}
}
\newcommand{\pipette}[2][]{

    \draw[double distance=1pt](0,0)--(#2,0);
    \draw[line join=round](#2,-0.7pt)--++(0.2,0.5pt)--++(0,0.4pt)--++(-0.2,0.5pt);

  \end{scope}
}
