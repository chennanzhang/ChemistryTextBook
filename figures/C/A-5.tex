\newcommand{\retortstand}[4][]{
  \begin{scope}[#1]
    \draw[fill=darkgray](0,0)--++(#3,0)--++(0.05,-0.05)--++(0,-0.15)--++(-0.4,0)--++(0,0.05)--++(-#2-#3+0.7,0)--++(0,-0.05)--++(-0.4,0)--++(0,0.15)--++(0.05,0.05)--cycle;
    \draw(-#2-0.05,-0.05)--(#3+0.05,-0.05);
    \fill[left color=lightgray,right color=lightgray,middle color=white](-0.05,0)rectangle(0.05,#4);
    \fill[left color=gray,right color=gray,middle color=white](-0.07,0)rectangle(0.07,0.07);
  \end{scope}
}
\newcommand{\beaker}[5][]{
  \begin{scope}[#1]
    \fill[#5,rounded corners=1mm](-#2/2,#4)--(-#2/2,0)--(#2/2,0)--(#2/2,#4);
    \draw[fill=cyan!10!white,fill opacity=0.3](#2/2+0.07,#3)arc(90:180:0.07)[rounded corners=1mm]--(#2/2,0)--(-#2/2,0)[rounded corners=0.5mm]--(-#2/2,#3-0.15)[sharp corners]--++(-0.1,0.1)--(-#2/2,#3)--cycle;
  \end{scope}
}

\newcommand{\UTube}[3][]{
  \begin{scope}[#1]
    \fill[cyan!40!white,fill opacity=0.5](#3:0.5)arc(#3:-#3-180:0.5)--cycle;
    \fill[left color=gray,right color=gray,middle color=white](-0.5,#2-0.1)rectangle++(0.3,0.25);
    \draw[fill=cyan!20!white,fill opacity=0.2](-0.5,0)arc(180:360:0.5)--(0.5,#2)arc(180:90:0.05)--++(-0.4,0)arc(90:0:0.05)--(0.2,0)arc(0:-180:0.2)--(-0.2,#2)arc(180:90:0.05)--++(-0.4,0)arc(90:0:0.05)--cycle;
  \end{scope}
}
\newcommand{\VTube}[3][]{

    \foreach \y in {-0.1,0,0.1,0.3}
    {\draw[very thin,fill=lightgray!20](0.1*rand,\y*rand)circle(0.8pt);}
    \draw[fill=cyan!20!white,fill opacity=0.2](-0.2,#2)arc(90:0:0.05)--(-0.15,0)arc(180:360:0.15)--(0.15,#2-0.05)arc(180:90:0.05)--cycle;
  \end{scope}
}
