\newcommand{\retortstand}[4][]{
  \begin{scope}[#1]
    \draw[fill=darkgray](0,0)--++(#3,0)--++(0.05,-0.05)--++(0,-0.15)--++(-0.4,0)--++(0,0.05)--++(-#2-#3+0.7,0)--++(0,-0.05)--++(-0.4,0)--++(0,0.15)--++(0.05,0.05)--cycle;
    \draw(-#2-0.05,-0.05)--(#3+0.05,-0.05);
    \fill[left color=lightgray,right color=lightgray,middle color=white](-0.05,0)rectangle(0.05,#4);
    \fill[left color=gray,right color=gray,middle color=white](-0.07,0)rectangle(0.07,0.07);
  \end{scope}
}
\newcommand{\rflask}[5][]{
  \begin{scope}[#1]
    \fill[#4,fill opacity=0.7](#3:0.5)arc(#3:-#3-180:0.5)--cycle;
    \fill[#5,fill opacity=0.3](#3:0.5)arc(#3:60:0.5)arc(240:180:0.2)--++(0,0.65)--++(-0.3,0)--++(0,-0.65)arc(0:-60:0.2)arc(120:180-#3:0.5);
    \fill[pattern={Dots[radius=0.2pt,distance=0.4pt,angle=45]}](-65:0.5)arc(-65:-115:0.5)to[bend left=10](-65:0.5);
    \draw(60:0.5)arc(60:-240:0.5)arc(-60:0:0.2)--++(0,0.5)coordinate(#2-bl)--++(0,0.1)arc(0:90:0.05)--++(0.4,0)arc(90:180:0.05)--++(0,-0.6)arc(180:240:0.2);
    \fill[left color=gray,right color=gray,middle color=white](#2-bl)rectangle++(0.3,0.25);
    \draw[semithick](-0.2,1.25)--(0.2,1.25);
  \end{scope}
}
\newcommand{\JointClipC}[3][]{
  \begin{scope}[#1]
    \fill[top color=gray,bottom color=gray,middle color=white](-0.2,-0.03)rectangle(#2,0.03);
    \fill[lightgray](#2,-0.07)rectangle++(#3,0.14);
    \fill[left color=darkgray,right color=darkgray,middle color=white](-0.1,-0.1)rectangle(0.1,0.1);
    \draw[line cap=round,very thick](-135:0.07)--(45:0.07);
    \fill[ball color=black](0,0)circle(1.4pt);
  \end{scope}
}
\newcommand{\ConicalFlask}[3][]{
  \begin{scope}[#1]
    \fill[#3,rounded corners=5pt](0.65-#2/3,#2)--(0.65,0)--(-0.65,0)--(-0.65+#2/3,#2);
    \draw[fill=cyan!5!white,fill opacity=0.2](-0.15,1.5)--(-0.15,1.8)arc(0:90:0.05)--(0.20,1.85)arc(90:180:0.05)[rounded corners=5pt]--(0.15,1.5)--(0.65,0)--(-0.65,0)--(-0.15,1.5)--cycle;
  \end{scope}
}
